kernel_image=uImage
devicetree_image=zynq-libresdr.dtb
kernel_load_address=0x3000000
devicetree_load_address=0x2A00000
bootargs=console=ttyPS0,115200 root=/dev/mmcblk0p2 rootfstype=squashfs ro rootwait
uenvcmd=mmcinfo && fatload mmc 0 ${kernel_load_address} ${kernel_image} && fatload mmc 0 ${devicetree_load_address} ${devicetree_image} && bootm ${kernel_load_address} - ${devicetree_load_address}
